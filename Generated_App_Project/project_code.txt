/**
 * نظام المؤثرات الصوتية باستخدام Web Audio API 
 * دون الحاجة لملفات صوتية خارجية
 */
class SoundEffects {
    constructor() {
        this.ctx = null;
        this.enabled = true;
    }

    init() {
        if (!this.ctx) {
            const AudioContext = window.AudioContext || window.webkitAudioContext;
            this.ctx = new AudioContext();
        }
        if (this.ctx.state === 'suspended') {
            this.ctx.resume();
        }
    }

    toggleSound() {
        this.enabled = !this.enabled;
        return this.enabled;
    }

    playEat() {
        if (!this.enabled) return;
        this.init();

        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(300, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(600, this.ctx.currentTime + 0.1);

        gain.gain.setValueAtTime(0.3, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.1);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        osc.start();
        osc.stop(this.ctx.currentTime + 0.1);
    }

    playBonus() {
        if (!this.enabled) return;
        this.init();

        const notes = [400, 500, 600, 800];
        notes.forEach((freq, index) => {
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();

            osc.type = 'triangle';
            osc.frequency.value = freq;

            const startTime = this.ctx.currentTime + index * 0.05;
            gain.gain.setValueAtTime(0.2, startTime);
            gain.gain.exponentialRampToValueAtTime(0.01, startTime + 0.08);

            osc.connect(gain);
            gain.connect(this.ctx.destination);

            osc.start(startTime);
            osc.stop(startTime + 0.08);
        });
    }

    playGameOver() {
        if (!this.enabled) return;
        this.init();

        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(250, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(80, this.ctx.currentTime + 0.4);

        gain.gain.setValueAtTime(0.3, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.4);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        osc.start();
        osc.stop(this.ctx.currentTime + 0.4);
    }

    playClick() {
        if (!this.enabled) return;
        this.init();

        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, this.ctx.currentTime);
        gain.gain.setValueAtTime(0.1, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.03);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        osc.start();
        osc.stop(this.ctx.currentTime + 0.03);
    }
}

/**
 * المحرك الرئيسي للعبة الثعبان
 */
document.addEventListener('DOMContentLoaded', () => {
    // شبكة اللعبة والكانفاس
    const canvas = document.getElementById('game-canvas');
    const ctx = canvas.getContext('2d');

    const GRID_SIZE = 20; // 20x20 مربعات
    let CELL_SIZE = canvas.width / GRID_SIZE;

    // عناصر القوائم والنصوص
    const currentScoreEl = document.getElementById('current-score');
    const highScoreEl = document.getElementById('high-score');
    const finalScoreValEl = document.getElementById('final-score-val');
    const newHighScoreMsg = document.getElementById('new-high-score-msg');

    // الشاشات
    const startOverlay = document.getElementById('start-overlay');
    const pauseOverlay = document.getElementById('pause-overlay');
    const gameoverOverlay = document.getElementById('gameover-overlay');
    const settingsModal = document.getElementById('settings-modal');

    // الأزرار
    const btnStart = document.getElementById('btn-start');
    const btnRestart = document.getElementById('btn-restart');
    const btnResume = document.getElementById('btn-resume');
    const btnPause = document.getElementById('btn-pause');
    const btnSound = document.getElementById('btn-sound');
    const btnSettings = document.getElementById('btn-settings');
    const btnCloseSettings = document.getElementById('btn-close-settings');

    // أدوات الإعدادات
    const selectDifficulty = document.getElementById('select-difficulty');
    const selectWalls = document.getElementById('select-walls');
    const selectTheme = document.getElementById('select-theme');

    // حالة اللعبة
    let snake = [];
    let food = { x: 0, y: 0 };
    let bonusFood = null;
    let bonusTimer = null;
    let direction = 'RIGHT';
    let nextDirection = 'RIGHT';
    let score = 0;
    let highScore = localStorage.getItem('snake_high_score') || 0;
    let gameLoop = null;
    let isPaused = false;
    let isGameOver = false;
    let isGameRunning = false;

    // سرعة اللعبة حسب الصعوبة (بالملي ثانية)
    const SPEED_MAP = {
        easy: 140,
        medium: 90,
        hard: 60
    };
    let currentSpeed = SPEED_MAP.medium;

    // تهيئة النتيجة القياسية العليا
    highScoreEl.textContent = highScore;

    // ضبط قياس الكانفاس بمرونة
    function resizeCanvas() {
        const rect = canvas.getBoundingClientRect();
        canvas.width = rect.width;
        canvas.height = rect.width;
        CELL_SIZE = canvas.width / GRID_SIZE;
        if (!isGameRunning) drawInitialGrid();
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    // بداية اللعبة
    function startGame() {
        snake = [
            { x: 5, y: 10 },
            { x: 4, y: 10 },
            { x: 3, y: 10 }
        ];
        direction = 'RIGHT';
        nextDirection = 'RIGHT';
        score = 0;
        isGameOver = false;
        isPaused = false;
        isGameRunning = true;
        currentScoreEl.textContent = '0';

        if (bonusTimer) clearTimeout(bonusTimer);
        bonusFood = null;

        spawnFood();
        
        startOverlay.classList.add('hidden');
        gameoverOverlay.classList.add('hidden');
        pauseOverlay.classList.add('hidden');
        btnPause.disabled = false;

        if (gameLoop) clearInterval(gameLoop);
        gameLoop = setInterval(update, currentSpeed);
    }

    // إنشاء الطعام
    function spawnFood() {
        food = {
            x: Math.floor(Math.random() * GRID_SIZE),
            y: Math.floor(Math.random() * GRID_SIZE)
        };
        // عدم إنتاج الطعام فوق جسم الثعبان
        while (snake.some(segment => segment.x === food.x && segment.y === food.y)) {
            food = {
                x: Math.floor(Math.random() * GRID_SIZE),
                y: Math.floor(Math.random() * GRID_SIZE)
            };
        }

        // فرصة 20% لإنشاء طعام مكافأة ذو وقت محدد
        if (Math.random() < 0.2 && !bonusFood) {
            spawnBonusFood();
        }
    }

    function spawnBonusFood() {
        bonusFood = {
            x: Math.floor(Math.random() * GRID_SIZE),
            y: Math.floor(Math.random() * GRID_SIZE)
        };
        bonusTimer = setTimeout(() => {
            bonusFood = null;
        }, 5000); // اختفاء المكافأة بعد 5 ثوانٍ
    }

    // تحديث حالة اللعبة (Game Loop Step)
    function update() {
        if (isPaused || isGameOver) return;

        direction = nextDirection;
        const head = { ...snake[0] };

        switch (direction) {
            case 'UP': head.y--; break;
            case 'DOWN': head.y++; break;
            case 'LEFT': head.x--; break;
            case 'RIGHT': head.x++; break;
        }

        // التعامل مع الجدران
        const wallMode = selectWalls.value;
        if (wallMode === 'solid') {
            if (head.x < 0 || head.x >= GRID_SIZE || head.y < 0 || head.y >= GRID_SIZE) {
                return gameOver();
            }
        } else { // المرور العابر
            if (head.x < 0) head.x = GRID_SIZE - 1;
            if (head.x >= GRID_SIZE) head.x = 0;
            if (head.y < 0) head.y = GRID_SIZE - 1;
            if (head.y >= GRID_SIZE) head.y = 0;
        }

        // الاصطدام بالنفس
        if (snake.some(segment => segment.x === head.x && segment.y === head.y)) {
            return gameOver();
        }

        snake.unshift(head);

        // أكل الطعام العادي
        if (head.x === food.x && head.y === food.y) {
            score += 10;
            currentScoreEl.textContent = score;
            audioFX.playEat();
            spawnFood();
        } 
        // أكل طعام المكافأة
        else if (bonusFood && head.x === bonusFood.x && head.y === bonusFood.y) {
            score += 30;
            currentScoreEl.textContent = score;
            audioFX.playBonus();
            bonusFood = null;
            clearTimeout(bonusTimer);
        } else {
            snake.pop(); // حركة طبيعية بدون زيادة للطول
        }

        render();
    }

    // رسم عناصر اللعبة على الكانفاس
    function render() {
        // مسح الكانفاس
        ctx.clearRect(0, 0, canvas.width, canvas.height);

        // رسم خلفية الشبكة
        ctx.strokeStyle = getComputedStyle(document.documentElement).getPropertyValue('--grid-line') || '#ffffff05';
        ctx.lineWidth = 1;
        for (let i = 0; i <= GRID_SIZE; i++) {
            ctx.beginPath();
            ctx.moveTo(i * CELL_SIZE, 0);
            ctx.lineTo(i * CELL_SIZE, canvas.height);
            ctx.stroke();

            ctx.beginPath();
            ctx.moveTo(0, i * CELL_SIZE);
            ctx.lineTo(canvas.width, i * CELL_SIZE);
            ctx.stroke();
        }

        // رسم طعام المكافأة (إن وجد)
        if (bonusFood) {
            ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue('--bonus-food-color') || '#fbbf24';
            ctx.shadowColor = '#fbbf24';
            ctx.shadowBlur = 10;
            ctx.beginPath();
            ctx.arc(
                bonusFood.x * CELL_SIZE + CELL_SIZE / 2,
                bonusFood.y * CELL_SIZE + CELL_SIZE / 2,
                CELL_SIZE / 2 - 2,
                0,
                Math.PI * 2
            );
            ctx.fill();
            ctx.shadowBlur = 0;
        }

        // رسم الطعام العادي
        ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue('--food-color') || '#ef4444';
        ctx.shadowColor = '#ef4444';
        ctx.shadowBlur = 8;
        ctx.beginPath();
        ctx.arc(
            food.x * CELL_SIZE + CELL_SIZE / 2,
            food.y * CELL_SIZE + CELL_SIZE / 2,
            CELL_SIZE / 2 - 3,
            0,
            Math.PI * 2
        );
        ctx.fill();
        ctx.shadowBlur = 0;

        // رسم الثعبان
        snake.forEach((segment, index) => {
            const isHead = index === 0;
            ctx.fillStyle = isHead 
                ? getComputedStyle(document.documentElement).getPropertyValue('--snake-head') || '#34d399'
                : getComputedStyle(document.documentElement).getPropertyValue('--snake-body') || '#10b981';

            const radius = isHead ? 6 : 4;
            const x = segment.x * CELL_SIZE;
            const y = segment.y * CELL_SIZE;

            // رسم مربع مستدير الحواف لجسم الثعبان
            ctx.beginPath();
            ctx.roundRect(x + 1, y + 1, CELL_SIZE - 2, CELL_SIZE - 2, radius);
            ctx.fill();

            // رسم عيني الثعبان للرأس
            if (isHead) {
                ctx.fillStyle = '#000';
                const eyeOffset = CELL_SIZE / 4;
                const eyeSize = 2;

                let leftEye = { x: x + eyeOffset, y: y + eyeOffset };
                let rightEye = { x: x + CELL_SIZE - eyeOffset, y: y + eyeOffset };

                if (direction === 'DOWN') {
                    leftEye = { x: x + eyeOffset, y: y + CELL_SIZE - eyeOffset };
                    rightEye = { x: x + CELL_SIZE - eyeOffset, y: y + CELL_SIZE - eyeOffset };
                } else if (direction === 'LEFT') {
                    leftEye = { x: x + eyeOffset, y: y + eyeOffset };
                    rightEye = { x: x + eyeOffset, y: y + CELL_SIZE - eyeOffset };
                } else if (direction === 'RIGHT') {
                    leftEye = { x: x + CELL_SIZE - eyeOffset, y: y + eyeOffset };
                    rightEye = { x: x + CELL_SIZE - eyeOffset, y: y + CELL_SIZE - eyeOffset };
                }

                ctx.beginPath();
                ctx.arc(leftEye.x, leftEye.y, eyeSize, 0, Math.PI * 2);
                ctx.arc(rightEye.x, rightEye.y, eyeSize, 0, Math.PI * 2);
                ctx.fill();
            }
        });
    }

    function drawInitialGrid() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
    }

    // عند انتهاء اللعبة
    function gameOver() {
        isGameOver = true;
        isGameRunning = false;
        clearInterval(gameLoop);
        audioFX.playGameOver();

        finalScoreValEl.textContent = score;

        if (score > highScore) {
            highScore = score;
            localStorage.setItem('snake_high_score', highScore);
            highScoreEl.textContent = highScore;
            newHighScoreMsg.classList.remove('hidden');
        } else {
            newHighScoreMsg.classList.add('hidden');
        }

        btnPause.disabled = true;
        gameoverOverlay.classList.remove('hidden');
    }

    // إيقاف واستئناف اللعبة
    function togglePause() {
        if (!isGameRunning || isGameOver) return;
        isPaused = !isPaused;
        audioFX.playClick();

        if (isPaused) {
            pauseOverlay.classList.remove('hidden');
            btnPause.querySelector('.icon').textContent = '▶';
        } else {
            pauseOverlay.classList.add('hidden');
            btnPause.querySelector('.icon').textContent = '⏸';
        }
    }

    // إدارة مفاتيح التوجيه (الكيبورد)
    function handleKeyPress(e) {
        if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(e.key)) {
            e.preventDefault();
        }

        if (e.key === ' ' || e.key === 'p' || e.key === 'P') {
            togglePause();
            return;
        }

        if (isPaused || !isGameRunning) return;

        switch (e.key) {
            case 'ArrowUp':
            case 'w':
            case 'W':
                if (direction !== 'DOWN') nextDirection = 'UP';
                break;
            case 'ArrowDown':
            case 's':
            case 'S':
                if (direction !== 'UP') nextDirection = 'DOWN';
                break;
            case 'ArrowLeft':
            case 'a':
            case 'A':
                if (direction !== 'RIGHT') nextDirection = 'LEFT';
                break;
            case 'ArrowRight':
            case 'd':
            case 'D':
                if (direction !== 'LEFT') nextDirection = 'RIGHT';
                break;
        }
    }

    // دعم السحب والإيماءات باللمس للشاشات الذكية (Swipe)
    let touchStartX = 0;
    let touchStartY = 0;

    canvas.addEventListener('touchstart', (e) => {
        touchStartX = e.touches[0].clientX;
        touchStartY = e.touches[0].clientY;
    }, { passive: true });

    canvas.addEventListener('touchend', (e) => {
        if (!isGameRunning || isPaused) return;

        let touchEndX = e.changedTouches[0].clientX;
        let touchEndY = e.changedTouches[0].clientY;

        let dx = touchEndX - touchStartX;
        let dy = touchEndY - touchStartY;

        if (Math.abs(dx) > Math.abs(dy)) {
            // سحب أفقي
            if (dx > 30 && direction !== 'LEFT') nextDirection = 'RIGHT';
            else if (dx < -30 && direction !== 'RIGHT') nextDirection = 'LEFT';
        } else {
            // سحب رأسي
            if (dy > 30 && direction !== 'UP') nextDirection = 'DOWN';
            else if (dy < -30 && direction !== 'DOWN') nextDirection = 'UP';
        }
    }, { passive: true });

    // ربط الأحداث بالأزرار
    document.addEventListener('keydown', handleKeyPress);

    btnStart.addEventListener('click', () => { audioFX.playClick(); startGame(); });
    btnRestart.addEventListener('click', () => { audioFX.playClick(); startGame(); });
    btnResume.addEventListener('click', togglePause);
    btnPause.addEventListener('click', togglePause);

    // ربط أزرار التحكم اللمسية (DPad)
    document.getElementById('btn-up').addEventListener('click', () => { if (direction !== 'DOWN') nextDirection = 'UP'; });
    document.getElementById('btn-down').addEventListener('click', () => { if (direction !== 'UP') nextDirection = 'DOWN'; });
    document.getElementById('btn-left').addEventListener('click', () => { if (direction !== 'RIGHT') nextDirection = 'LEFT'; });
    document.getElementById('btn-right').addEventListener('click', () => { if (direction !== 'LEFT') nextDirection = 'RIGHT'; });

    // زر التحكم بالصوت
    btnSound.addEventListener('click', () => {
        const enabled = audioFX.toggleSound();
        btnSound.querySelector('.icon').textContent = enabled ? '🔊' : '🔇';
    });

    // الإعدادات والمودال
    btnSettings.addEventListener('click', () => {
        audioFX.playClick();
        if (isGameRunning && !isPaused) togglePause();
        settingsModal.classList.remove('hidden');
    });

    btnCloseSettings.addEventListener('click', () => {
        audioFX.playClick();
        settingsModal.classList.add('hidden');
    });

    selectDifficulty.addEventListener('change', (e) => {
        currentSpeed = SPEED_MAP[e.target.value];
        if (isGameRunning) {
            clearInterval(gameLoop);
            gameLoop = setInterval(update, currentSpeed);
        }
    });

    selectTheme.addEventListener('change', (e) => {
        document.body.className = '';
        if (e.target.value !== 'neon') {
            document.body.classList.add(`theme-${e.target.value}`);
        }
        render();
    });
});

### 🌟 مميزات التطبيق:
1. **واجهة باللغة العربية (RTL):** تصميم وتنسيق متوافق مع الاتجاه من اليمين إلى اليسار واستخدام خط `Cairo` الأنيق.
2. **تصميم متجاوب عصري (Glassmorphism):** يعمل على كافة الشاشات (الهواتف المحمولة، التابلت، والحواسيب).
3. **أزرار تحكم متكاملة للهواتف:** لوحة أزرار افتراضية (D-Pad) + دعم الإيماءات والسحب باللمس (Swipe gestures).
4. **تأثيرات صوتية تفاعلية (Web Audio API):** أصوات مدمجة برمجياً للأكل، المكافآت، الخسارة، والأزرار دون الحاجة لملفات externe MP3.
5. **مستويات صعوبة وأنماط لعب متعددة:**
   - خيارات الصعوبة (سهل، متوسط، صعب).
   - نمط الجدران (صلبة تؤدي للموت أو عبور ينقل للجانب الآخر).
   - ثيمات ألوان متعددة (نيون، كلاسيكي أخضر، سايبربانك).
6. **حفظ أعلى نتيجة (High Score):** حفظ تلقائي في ذاكرة المتصفح `localStorage`.
